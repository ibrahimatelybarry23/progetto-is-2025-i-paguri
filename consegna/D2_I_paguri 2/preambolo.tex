% --- Font -------------------------------------------------------------
% Il Word usa Calibri (testo) e Cambria (voci dell'indice).
% Se non sono installati si usano Carlito e Caladea, che hanno
% le stesse identiche metriche (stessa larghezza dei caratteri).
\IfFontExistsTF{Calibri}
  {\setmainfont{Calibri}}
  {\setmainfont{Carlito}[Extension=.ttf, UprightFont=*-Regular,
     BoldFont=*-Bold, ItalicFont=*-Italic, BoldItalicFont=*-BoldItalic]}

  {\newfontfamily\cambria{Cambria}}
  {\newfontfamily\cambria{Caladea}[Extension=.ttf, UprightFont=*-Regular,
     BoldFont=*-Bold, ItalicFont=*-Italic, BoldItalicFont=*-BoldItalic]}
